cmake_minimum_required(VERSION 3.22.1)
project(audioassistant LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Only arm64-v8a is supported")
endif()
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(AUDIO_NATIVE_PREFIX "${REPOSITORY_ROOT}/.native/prefix/${ANDROID_ABI}" CACHE PATH "Real locally built FFmpeg/LAME prefix")
foreach(component avformat avcodec avutil swresample avfilter mp3lame)
    if(NOT EXISTS "${AUDIO_NATIVE_PREFIX}/lib/lib${component}.so")
        message(FATAL_ERROR "Missing real lib${component}.so; run scripts/build-native.sh locally; see docs/LOCAL_VALIDATION.md")
    endif()
    add_library(${component} SHARED IMPORTED)
    set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${AUDIO_NATIVE_PREFIX}/lib/lib${component}.so")
endforeach()
add_library(audioengine SHARED audio_engine.cpp)
target_include_directories(audioengine PRIVATE "${AUDIO_NATIVE_PREFIX}/include")
target_compile_options(audioengine PRIVATE -Wall -Wextra -fvisibility=hidden)
target_link_options(audioengine PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(audioengine PRIVATE avformat avcodec avutil swresample avfilter mp3lame)
